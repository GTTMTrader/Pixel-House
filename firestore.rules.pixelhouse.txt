rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function isMember() {
      return request.auth != null
        && request.auth.token.email in ['hellin@study.com', 'moon@study.com'];
    }
    match /pixelhouse/{houseId}/{document=**} {
      allow read, write: if isMember();
    }
  }
}
